│   ├── admin_users.html   # Zarządzanie użytkownikami (lista, dodawanie, edycja, usuwanie) – tylko admin
│   ├── user_form.html     # Formularz dodawania/edycji użytkownika (rola, nieobecności, hasło)
│   ├── archive.html       # Archiwum projektów (przywracanie lub trwałe usuwanie) – tylko admin
│   ├── calendar.html      # Kalendarz z terminami projektów i nieobecnościami użytkowników (FullCalendar)
│   ├── charts.html        # Strona z wykresami: przepływ zadań (ostatnie 30 dni), obciążenie pracowników, wąskie gardła, zadania oczekujące wg roli, planowane vs rzeczywiste (dla ukończonych zadań)
│   ├── logs.html          # Dziennik zdarzeń systemowych (paginacja, automatyczne usuwanie wpisów starszych niż 3 miesiące) – tylko admin
│   ├── kanban.html        # Tablica Kanban (kolumny: Do zrobienia, W trakcie/Wstrzymane, Zrobione) z przeciąganiem elementów (SortableJS) – aktualizacja statusu przez API
│   ├── tasks_pdf.html     # Szablon HTML dla raportów PDF (moje zadania lub raport zespołu) – używany przez WeasyPrint
│   ├── admin_project_templates.html   # Lista szablonów projektów (zarządzanie: dodawanie, edycja, usuwanie) – tylko admin
│   ├── project_template_form.html     # Formularz dodawania/edycji szablonu projektu (oraz zarządzanie jego krokami)
│   ├── admin_step_templates.html      # Lista samodzielnych szablonów kroków (zarządzanie) – tylko admin
│   └── step_template_form.html        # Formularz dodawania/edycji szablonu kroku (samodzielnego lub w ramach szablonu projektu)
└── static/
    └── style.css           # Własne style: szare tło, karty, paski postępu, kolory statusów (todo/in_progress/done), responsywność, style dla Kanban, toasty (choć system powiadomień wyłączony), wykresy
