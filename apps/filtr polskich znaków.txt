unit Unit1;

{$mode objfpc}{$H+}

interface

uses
  Classes, SysUtils, Forms, Controls, Graphics, Dialogs, StdCtrls;

type

  { TForm1 }

  TForm1 = class(TForm)
    Edit1: TEdit;
    procedure Edit1Change(Sender: TObject; var Key: char);
  private
    function RemovePolishChars(const InputStr: String): String;

  public

  end;

var
  Form1: TForm1;

implementation

{$R *.lfm}

{ TForm1 }

procedure TForm1.Edit1Change(Sender: TObject; var Key: char);
var
  FilteredText: String;
begin
  // Usuwanie polskich znaków z tekstu wprowadzonego do pola edycji
  FilteredText := RemovePolishChars(Edit1.Text);
  // Ustawienie filtrowanego tekstu z powrotem do pola edycji
  Edit1.Text := FilteredText;
  // Ustawienie kursora na koniec tekstu
  Edit1.SelStart := Length(FilteredText);

end;

function TForm1.RemovePolishChars(const InputStr: String): String;
const
  PolishChars = 'ąćęłńóśźżĄĆĘŁŃÓŚŹŻ';
var
  i: Integer;
  CharIndex: Integer;
begin
  Result := InputStr;
  for i := 1 to Length(PolishChars) do
  begin
    CharIndex := Pos(PolishChars[i], Result);
    while CharIndex > 0 do
    begin
      Delete(Result, CharIndex, 1);
      CharIndex := Pos(PolishChars[i], Result);
    end;
  end;
end;

end.
